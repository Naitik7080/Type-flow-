@Composable
fun KisanSathiApp() {
    val green = Color(0xFF2E7D32)
    var selectedTab by remember { mutableIntStateOf(0) }
    var dialogTitle by remember { mutableStateOf("") }
    var dialogMessage by remember { mutableStateOf("") }
    var showDialog by remember { mutableStateOf(false) }

    fun showInfo(title: String, message: String) {
        dialogTitle = title
        dialogMessage = message
        showDialog = true
    }

    Scaffold(
        topBar = {
            TopAppBar(
                title = {
                    Column {
                        Text("Kisan Sathi", fontWeight = FontWeight.Bold)
                        Text("Your farming companion", fontSize = 12.sp)
                    }
                }
            )
        },
        bottomBar = {
            NavigationBar {
                NavigationBarItem(
                    selected = selectedTab == 0,
                    onClick = { selectedTab = 0 },
                    icon = { Text("🏠") },
                    label = { Text("Home") }
                )
                NavigationBarItem(
                    selected = selectedTab == 1,
                    onClick = { selectedTab = 1 },
                    icon = { Text("🌾") },
                    label = { Text("Services") }
                )
                NavigationBarItem(
                    selected = selectedTab == 2,
                    onClick = { selectedTab = 2 },
                    icon = { Text("🤝") },
                    label = { Text("Support") }
                )
                NavigationBarItem(
                    selected = selectedTab == 3,
                    onClick = { selectedTab = 3 },
                    icon = { Text("ℹ️") },
                    label = { Text("About") }
                )
            }
        }
    ) { padding ->
        when (selectedTab) {
            0 -> HomeScreen(padding, green, ::showInfo)
            1 -> ServicesScreen(padding, green, ::showInfo)
            2 -> SupportScreen(padding, green, ::showInfo)
            3 -> AboutScreen(padding, green)
        }
    }

    if (showDialog) {
        AlertDialog(
            onDismissRequest = { showDialog = false },
            title = { Text(dialogTitle, fontWeight = FontWeight.Bold) },
            text = { Text(dialogMessage) },
            confirmButton = {
                Button(onClick = { showDialog = false }) { Text("OK") }
            }
        )
    }
}

@Composable
fun ServicesScreen(
    padding: PaddingValues,
    green: Color,
    onInfo: (String, String) -> Unit
) {
    val services = listOf(
        Triple("Government Schemes", "Government schemes for farmers", "🏛️"),
        Triple("Mandi Rates", "Check crop market prices", "💰"),
        Triple("Weather", "Weather information", "☁️"),
        Triple("Crop Information", "Crop growing information", "🌱"),
        Triple("Farming Guidance", "Useful farming tips", "💡"),
        Triple("Kisan Support", "Help farmers in need", "🤝")
    )

    LazyColumn(
        modifier = Modifier
            .fillMaxSize()
            .background(Color(0xFFF7F9F7))
            .padding(padding),
        contentPadding = PaddingValues(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        item {
            Text("Farmer Services 🌾", fontSize = 25.sp, fontWeight = FontWeight.Bold)
            Spacer(Modifier.height(4.dp))
            Text("Useful services for farmers")
        }

        items(services) { service ->
            ServiceCard(service.first, service.second, service.third, green) {
                onInfo(service.first, service.second)
            }
        }
    }
}

@Composable

@Composable
fun InfoCard(
    modifier: Modifier,
    emoji: String,
    title: String,
    value: String,
    color: Color,
    onClick: () -> Unit
) {
    Card(
        modifier = modifier.clickable { onClick() },
        shape = RoundedCornerShape(18.dp),
        colors = CardDefaults.cardColors(containerColor = Color.White),
        elevation = CardDefaults.cardElevation(3.dp)
    ) {
        Column(Modifier.padding(16.dp)) {
            Text(emoji, fontSize = 30.sp)
            Spacer(Modifier.height(10.dp))
            Text(title, fontWeight = FontWeight.Bold)
            Text(value, color = color, fontSize = 14.sp)
        }
    }
}

@Composable

    subtitle: String,

) {

        modifier = Modifier
            .fillMaxWidth()
            .clickable { onClick() },
        shape = RoundedCornerShape(18.dp),
        colors = CardDefaults.cardColors(containerColor = Color.White),
        elevation = CardDefaults.cardElevation(2.dp)
    ) {
        Row(
            Modifier
                .fillMaxWidth()
                .padding(16.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            Box(
                modifier = Modifier
                    .size(52.dp)
                    .background(Color(0xFFE8F5E9), RoundedCornerShape(15.dp)),

                Text(emoji, fontSize = 27.sp)

            Spacer(Modifier.width(14.dp))

            Column(Modifier.weight(1f)) {
                Text(title, fontSize = 17.sp, fontWeight = FontWeight.Bold)
                Text(subtitle, fontSize = 13.sp, color = Color.Gray)
