// ============================================================

data class Tx(val en: String, val hi: String)

val LocalHindi = compositionLocalOf { false }

@Composable
fun Tx.t(): String = if (LocalHindi.current) hi else en

@Composable
fun tr(en: String, hi: String): String = if (LocalHindi.current) hi else en

// ============================================================
//  PREFERENCES  (language, theme, saved mandi items)
// ============================================================
class AppPrefs(context: Context) {
    private val sp = context.getSharedPreferences("kisan_sathi_prefs", Context.MODE_PRIVATE)

    var isHindi: Boolean
        get() = sp.getBoolean("hindi", false)
        set(v) { sp.edit().putBoolean("hindi", v).apply() }

    /** 0 = system, 1 = light, 2 = dark */
    var themeMode: Int
        get() = sp.getInt("theme", 0)
        set(v) { sp.edit().putInt("theme", v).apply() }

    var favorites: Set<String>
        get() = HashSet(sp.getStringSet("fav", emptySet()) ?: emptySet())
        set(v) { sp.edit().putStringSet("fav", HashSet(v)).apply() }
}

data class Crop(

    val tempC: Int,
    val condition: Tx,
    val humidity: Int,
    val windKmh: Int,
    val rainChance: Int,

fun weatherAdvisory(w: Weather): Tx = when {
    w.rainChance >= 60 -> Tx(
        "Rain likely today. Postpone spraying and fertilizer; keep field drains clear.",
        "आज बारिश की संभावना है। छिड़काव और खाद टालें; खेत की नालियां साफ रखें।"
    )
    w.windKmh >= 25 -> Tx(
        "Strong winds. Avoid pesticide spraying today.",
        "तेज़ हवा है। आज कीटनाशक का छिड़काव न करें।"
    )
    w.tempC >= 38 -> Tx(
        "Very hot. Irrigate in the evening and protect young plants.",
        "बहुत गर्मी है। शाम को सिंचाई करें और छोटे पौधों को बचाएं।"
    )
    else -> Tx(
        "Good conditions for field work. Spray in early morning or evening.",
        "खेत के काम के लिए अच्छा मौसम। छिड़काव सुबह जल्दी या शाम को करें।"
    )
}

sealed interface UiState<out T> {
    object Loading : UiState<Nothing>
    data class Success<T>(val data: T) : UiState<T>
    data class Error(val message: String) : UiState<Nothing>
}

@Composable
fun <T> loadState(reloadKey: Int, loader: suspend () -> T): UiState<T> {
    var state by remember(reloadKey) { mutableStateOf<UiState<T>>(UiState.Loading) }
    LaunchedEffect(reloadKey) {
        state = try {
            UiState.Success(loader())
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            UiState.Error(e.message ?: "Error")
        }
    }
    return state
}

enum class AppTab(val emoji: String, val label: Tx) {
    HOME("🏠", Tx("Home", "होम")),
    MANDI("💰", Tx("Mandi", "मंडी")),
    CROPS("🌱", Tx("Crops", "फसलें")),
    SCHEMES("🏛️", Tx("Schemes", "योजनाएं")),
    MORE("☰", Tx("More", "और"))
}

sealed interface Detail {
    data class CropD(val crop: Crop) : Detail
    data class SchemeD(val scheme: Scheme) : Detail
    object Calculator : Detail
    object Support : Detail
    object About : Detail
}

private val rupeeFormat: NumberFormat = NumberFormat.getInstance(Locale("en", "IN"))

fun rupee(n: Number): String {
    val v = n.toDouble().toLong()
    return (if (v < 0) "-₹" else "₹") + rupeeFormat.format(abs(v))
}

fun Context.dial(number: String) {
    runCatching { startActivity(Intent(Intent.ACTION_DIAL, Uri.parse("tel:$number"))) }
}

fun Context.openUrl(url: String) {
    runCatching { startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(url))) }
}

fun Context.shareText(text: String) {
    val i = Intent(Intent.ACTION_SEND).apply {
        type = "text/plain"
        putExtra(Intent.EXTRA_TEXT, text)
    }
    runCatching { startActivity(Intent.createChooser(i, null)) }
}

fun Context.emailUs(to: String, subject: String) {
    val i = Intent(Intent.ACTION_SENDTO).apply {
        data = Uri.parse("mailto:$to")
        putExtra(Intent.EXTRA_SUBJECT, subject)
    }
    runCatching { startActivity(i) }
}

@Composable
fun KisanSathiApp() {
    val context = LocalContext.current
    val prefs = remember { AppPrefs(context) }
    val repo: FarmRepository = remember { DemoFarmRepository() } // TODO: swap with ApiFarmRepository()

    var hindi by remember { mutableStateOf(prefs.isHindi) }
    var themeMode by remember { mutableStateOf(prefs.themeMode) }
    var favorites by remember { mutableStateOf(prefs.favorites) }
    var tabIndex by rememberSaveable { mutableStateOf(0) }
    var detail by remember { mutableStateOf<Detail?>(null) }
    var reloadKey by rememberSaveable { mutableStateOf(0) }

    val weatherState = loadState(reloadKey) { repo.weather() }
    val mandiState = loadState(reloadKey) { repo.mandiRates() }

    val dark = when (themeMode) {
        1 -> false
        2 -> true
        else -> isSystemInDarkTheme()
    }

    val tab = AppTab.values()[tabIndex]
    val toggleFav: (String) -> Unit = { id ->
        favorites = if (id in favorites) favorites - id else favorites + id
        prefs.favorites = favorites
    }

    BackHandler(enabled = detail != null) { detail = null }
    BackHandler(enabled = detail == null && tab != AppTab.HOME) { tabIndex = 0 }

    CompositionLocalProvider(LocalHindi provides hindi) {
        KisanTheme(dark) {
            val cs = MaterialTheme.colorScheme
            Scaffold(
                topBar = {
                    TopAppBar(
                        title = {
                            val d = detail
                            if (d == null) {
                                Column {
                                    Text("Kisan Sathi", fontWeight = FontWeight.Bold)
                                    Text(tr("Your farming companion", "आपका खेती का साथी"), fontSize = 12.sp)
                                }
                            } else {
                                Text(
                                    when (d) {
                                        is Detail.CropD -> d.crop.name.t()
                                        is Detail.SchemeD -> d.scheme.name.t()
                                        Detail.Calculator -> tr("Profit Calculator", "मुनाफ़ा कैलकुलेटर")
                                        Detail.Support -> tr("Support a Farmer", "किसान की मदद करें")
                                        Detail.About -> tr("About", "जानकारी")
                                    },
                                    fontWeight = FontWeight.Bold, maxLines = 1, overflow = TextOverflow.Ellipsis
                                )
                            }
                        },
                        navigationIcon = {
                            if (detail != null) {
                                IconButton(onClick = { detail = null }) {
                                    Text("←", color = cs.onPrimary, fontSize = 24.sp)
                                }
                            }
                        },
                        actions = {
                            TextButton(onClick = { hindi = !hindi; prefs.isHindi = hindi }) {
                                Text(if (hindi) "EN" else "हिं", color = cs.onPrimary, fontWeight = FontWeight.Bold)
                            }
                        },
                        colors = TopAppBarDefaults.topAppBarColors(
                            containerColor = cs.primary,
                            titleContentColor = cs.onPrimary,
                            navigationIconContentColor = cs.onPrimary,
                            actionIconContentColor = cs.onPrimary
                        )
                    )
                },
                bottomBar = {
                    if (detail == null) {
                        NavigationBar {
                            AppTab.values().forEach { t ->
                                NavigationBarItem(
                                    selected = tab == t,
                                    onClick = { tabIndex = t.ordinal },
                                    icon = { Text(t.emoji, fontSize = 20.sp) },
                                    label = { Text(t.label.t(), maxLines = 1, fontSize = 11.sp) }
                                )
                            }
                        }
                    }
                }
            ) { padding ->
                Box(
                    Modifier
                        .fillMaxSize()
                        .background(cs.background)
                        .padding(padding)
                ) {
                    val d = detail
                    if (d != null) {
                        when (d) {
                            is Detail.CropD -> {
                                val price = (mandiState as? UiState.Success)?.data
                                    ?.firstOrNull { it.cropId == d.crop.id }
                                CropDetail(d.crop, price, repo.isDemo)
                            }
                            is Detail.SchemeD -> SchemeDetail(d.scheme)
                            Detail.Calculator -> CalculatorScreen()
                            Detail.Support -> SupportScreen()
                            Detail.About -> AboutScreen()
                        }
                    } else {
                        when (tab) {
                            AppTab.HOME -> HomeScreen(
                                weatherState, mandiState, repo.isDemo,
                                onRetry = { reloadKey++ },
                                goTab = { tabIndex = it.ordinal },
                                open = { detail = it }
                            )
                            AppTab.MANDI -> MandiScreen(
                                mandiState, favorites, repo.isDemo,
                                onFav = toggleFav, onRetry = { reloadKey++ }
                            )
                            AppTab.CROPS -> CropsScreen(open = { detail = it })
                            AppTab.SCHEMES -> SchemesScreen(open = { detail = it })
                            AppTab.MORE -> MoreScreen(
                                themeMode = themeMode,
                                onHindi = { hindi = it; prefs.isHindi = it },
                                hindi = hindi,
                                onTheme = { themeMode = it; prefs.themeMode = it },
                                open = { detail = it }
                            )
                        }
                    }
                }
            }
        }
    }
}

@Composable
fun AppCard(
    modifier: Modifier = Modifier,
    onClick: (() -> Unit)? = null,
    container: Color = MaterialTheme.colorScheme.surface,
    content: @Composable ColumnScope.() -> Unit
) {
    val shape = RoundedCornerShape(18.dp)
    val colors = CardDefaults.cardColors(containerColor = container)
    val elevation = CardDefaults.cardElevation(defaultElevation = 2.dp)
    if (onClick != null) {
        Card(
            onClick = onClick, modifier = modifier.fillMaxWidth(), shape = shape,
            colors = colors, elevation = elevation, content = content
        )
    } else {
        Card(
            modifier = modifier.fillMaxWidth(), shape = shape,
            colors = colors, elevation = elevation, content = content
        )
    }
}

@Composable
fun TrendChip(changePct: Double) {
    val up = changePct >= 0
    val color = if (up) Color(0xFF2E7D32) else Color(0xFFC62828)
    Text(
        (if (up) "▲ " else "▼ ") + String.format(Locale.US, "%.1f%%", abs(changePct)),
        color = color, fontSize = 12.sp, fontWeight = FontWeight.Bold
    )
}

@Composable
fun MandiScreen(
    state: UiState<List<MandiRate>>,
    favorites: Set<String>,
    isDemo: Boolean,
    onFav: (String) -> Unit,
    onRetry: () -> Unit
) {
    var query by rememberSaveable { mutableStateOf("") }
    var filter by rememberSaveable { mutableStateOf("ALL") }
    val searchHint = tr("Search crop, mandi or state", "फसल, मंडी या राज्य खोजें")
    val allLabel = tr("All", "सभी")
    val savedLabel = tr("★ Saved", "★ सेव्ड")
    val emptyText = tr("No rates found.", "कोई भाव नहीं मिला।")

    LazyColumn(
        Modifier.fillMaxSize(),
        contentPadding = PaddingValues(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        item { SearchField(query, { query = it }, searchHint) }

        when (state) {
            is UiState.Loading -> item { LoadingBox() }
            is UiState.Error -> item { ErrorBox(onRetry) }
            is UiState.Success -> {
                val states = state.data.map { it.state }.distinct().sorted()
                val options = listOf("ALL" to allLabel, "FAV" to savedLabel) + states.map { it to it }
                item { ChipRow(options, filter) { filter = it } }
                if (isDemo) item { DemoBanner() }

                val q = query.trim().lowercase()
                val list = state.data.filter { r ->
                    val crop = Catalog.cropById(r.cropId)
                    val matchQ = q.isEmpty() ||
                        r.mandi.lowercase().contains(q) ||
                        r.state.lowercase().contains(q) ||
                        (crop?.name?.en?.lowercase()?.contains(q) == true) ||
                        (crop?.name?.hi?.contains(q) == true)
                    val matchF = when (filter) {
                        "ALL" -> true
                        "FAV" -> r.id in favorites
                        else -> r.state == filter
                    }
                    matchQ && matchF
                }

                if (list.isEmpty()) {
                    item {
                        Text(emptyText, Modifier.fillMaxWidth().padding(24.dp), textAlign = TextAlign.Center,
                            color = MaterialTheme.colorScheme.onSurfaceVariant)
                    }
                } else {
                    items(list, key = { it.id }) { r ->
                        MandiCard(r, fav = r.id in favorites, onFav = { onFav(r.id) })
                    }
                }
            }
        }
    }
}

@Composable
fun CalculatorScreen() {
    var area by rememberSaveable { mutableStateOf("") }
    var yieldQ by rememberSaveable { mutableStateOf("") }
    var price by rememberSaveable { mutableStateOf("") }
    var cost by rememberSaveable { mutableStateOf("") }

    val a = area.toDoubleOrNull()
    val y = yieldQ.toDoubleOrNull()
    val p = price.toDoubleOrNull()
    val c = cost.toDoubleOrNull()

    Column(
        Modifier.fillMaxSize().verticalScroll(rememberScrollState()).padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        Text(
            tr("Enter your numbers to estimate profit for one crop.", "एक फसल का मुनाफ़ा जानने के लिए अपने आंकड़े भरें।"),
            color = MaterialTheme.colorScheme.onSurfaceVariant
        )
        NumberField(area, { area = it }, tr("Land area (acres)", "ज़मीन (एकड़)"))
        NumberField(yieldQ, { yieldQ = it }, tr("Expected yield (quintal per acre)", "अनुमानित उपज (क्विंटल प्रति एकड़)"))
        NumberField(price, { price = it }, tr("Selling price (₹ per quintal)", "बिक्री भाव (₹ प्रति क्विंटल)"))
        NumberField(cost, { cost = it }, tr("Cost of farming (₹ per acre)", "खेती की लागत (₹ प्रति एकड़)"))

        if (a != null && y != null && p != null && c != null && a > 0) {
            val totalYield = a * y
            val revenue = totalYield * p
            val totalCost = a * c
            val profit = revenue - totalCost
            val perAcre = profit / a
            val breakEven = if (totalYield > 0) totalCost / totalYield else 0.0
            val good = profit >= 0
            val profitColor = if (good) Color(0xFF2E7D32) else Color(0xFFC62828)

            AppCard(container = MaterialTheme.colorScheme.primaryContainer) {
                Column(Modifier.padding(16.dp), verticalArrangement = Arrangement.spacedBy(10.dp)) {
                    Text(tr("Estimate", "अनुमान"), fontWeight = FontWeight.Bold, fontSize = 18.sp)
                    ResultLine(tr("Total production", "कुल उत्पादन"), String.format(Locale.US, "%.1f", totalYield) + tr(" qtl", " क्विं."))
                    ResultLine(tr("Total income", "कुल आमदनी"), rupee(revenue))
                    ResultLine(tr("Total cost", "कुल लागत"), rupee(totalCost))
                    HorizontalDivider()
                    ResultLine(
                        if (good) tr("Net profit", "शुद्ध मुनाफ़ा") else tr("Net loss", "शुद्ध नुकसान"),
                        rupee(profit), bold = true, color = profitColor
                    )
                    ResultLine(tr("Profit per acre", "मुनाफ़ा प्रति एकड़"), rupee(perAcre))
                    ResultLine(tr("Break-even price", "ब्रेक-ईवन भाव"), rupee(breakEven) + tr("/qtl", "/क्विं."))
                }
            }
            Text(
                tr("This is only an estimate. Actual results depend on weather, market price and expenses.",
                    "यह केवल अनुमान है। असली नतीजे मौसम, बाज़ार भाव और खर्च पर निर्भर करते हैं।"),
                fontSize = 12.sp, color = MaterialTheme.colorScheme.onSurfaceVariant
            )
        }
    }
}
